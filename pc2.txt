def connect_tcp():
    global connected
    target_ip = ip_entry.get().strip()

# ==========================================================
# TCP CLIENT RECEIVE THREAD (NH?N D? LI?U T? PC1)
# ==========================================================

def tcp_client_thread(ip, port):
    global connected, tcp_socket

    try:
        add_log(f"Ðang k?t n?i t?i PC1 ({ip}:{port})...")
        tcp_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        tcp_socket.settimeout(5.0)
        tcp_socket.connect((ip, port))
        
        connected = True
        gui_call(update_connection_ui_connected)

        buffer = ""
        tcp_socket.settimeout(3.0)

        while running and connected:
            try:
                data = tcp_socket.recv(1024).decode("utf-8")
                if not data:
                    break
                
                buffer += data
                while "\n" in buffer:
                    line, buffer = buffer.split("\n", 1)
                    if line.strip():
                        packet = json.loads(line.strip())
                        gui_call(lambda p=packet: process_incoming_packet(p))

            except socket.timeout:
                continue
            except Exception as e:
                add_log(f"L?i nh?n d? li?u: {e}", "ERROR")
                break

    except Exception as e:
        add_log(f"Không th? k?t n?i PC1 ({ip}:{port}): {e}", "ERROR")
        gui_call(lambda: show_toast(" L?i k?t n?i PC1!", RED))

    finally:
        connected = False
        if tcp_socket:
            try:
                tcp_socket.close()
            except Exception:
                pass
            tcp_socket = None
        gui_call(update_connection_ui_disconnected)


# ==========================================================
# PROCESS INCOMING PACKET
# ==========================================================

def process_incoming_packet(packet):
    global temperature, humidity, fan_state, pump_state

    temperature = float(packet.get("temperature", 0.0))
    humidity = float(packet.get("humidity", 0.0))
    sensor_online = packet.get("sensor_online", False)

    # AUTO MODE LOGIC
    if mode == "AUTO":
        fan_state = (temperature >= temp_threshold)
        pump_state = (humidity <= humi_threshold)
        update_device_ui()

    # UPDATE DASHBOARD VALUES
    dashboard_temp.configure(text=f"{temperature:.1f} °C")
    dashboard_humi.configure(text=f"{humidity:.1f} %")

    # UPDATE MONITOR VALUES
    monitor_temp.configure(text=f"{temperature:.1f} °C")
    monitor_humi.configure(text=f"{humidity:.1f} %")

    # ALARM CONDITIONS
    if temperature >= temp_threshold:
        temp_condition.configure(text="? NHI?T Ð? CAO", text_color=ORANGE)
    else:
        temp_condition.configure(text="? BÌNH THU?NG", text_color=GREEN)

    if humidity <= humi_threshold:
        humi_condition.configure(text="? Ð? ?M TH?P", text_color=ORANGE)
    else:
        humi_condition.configure(text="? BÌNH THU?NG", text_color=GREEN)

    # HISTORY RECORDING
    now_str = datetime.now().strftime("%H:%M:%S")
    temp_history.append(temperature)
    humi_history.append(humidity)
    time_history.append(now_str)

    if len(temp_history) > 30:
        temp_history.pop(0)
        humi_history.pop(0)
        time_history.pop(0)

    record = (
        datetime.now().strftime("%d/%m/%Y %H:%M:%S"),
        f"{temperature:.1f}",
        f"{humidity:.1f}",
        "ON" if fan_state else "OFF",
        "ON" if pump_state else "OFF",
        mode
    )

    history_records.append(record)
    update_history_table()
    update_charts()
    update_statistics()

def update_history_table():
    if not history_records:
        return

    record = history_records[-1]
    history_tree.insert("", 0, values=record)
    children = history_tree.get_children()

    if len(children) > 100:
        history_tree.delete(children[-1])


def clear_history():
    for item in history_tree.get_children():
        history_tree.delete(item)
    history_records.clear()
    show_toast("Ðã xóa l?ch s? d? li?u", ORANGE)


def export_history():
    if not history_records:
        messagebox.showwarning("L?ch s?", "Chua có d? li?u d? xu?t file.")
        return

    filename = filedialog.asksaveasfilename(
        defaultextension=".csv",
        filetypes=[("CSV file", "*.csv")]
    )

    if not filename:
        return

    with open(filename, "w", newline="", encoding="utf-8-sig") as file:
        writer = csv.writer(file)
        writer.writerow(["Th?i gian", "Nhi?t d? (°C)", "Ð? ?m (%)", "Qu?t", "Bom", "Ch? d?"])
        writer.writerows(history_records)

    show_toast(" Xu?t CSV thành công", GREEN)
